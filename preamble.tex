% preamble.tex — use Open Sans static TTFs for PDF via XeLaTeX
\usepackage{fontspec}

% Make LaTeX use Unicode/OpenType fonts:
  \defaultfontfeatures{Ligatures=TeX}

% Main (serif/sans) text face — Open Sans is a sans-serif, which is fine for books too.
% We explicitly map all faces you have so bold/italic resolve correctly.
\setmainfont[
  Path            = fonts/,
  Extension       = .ttf,
  UprightFont     = OpenSans-Regular.ttf,
  ItalicFont      = OpenSans-Italic.ttf,
  BoldFont        = OpenSans-Bold.ttf,
  BoldItalicFont  = OpenSans-BoldItalic.ttf
]{Open Sans}
